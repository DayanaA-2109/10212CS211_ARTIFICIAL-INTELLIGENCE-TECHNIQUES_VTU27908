PROGRAM :

import random

# Travel duration between locations
distance = [
    [0, 10, 15, 20],
    [10, 0, 35, 25],
    [15, 35, 0, 30],
    [20, 25, 30, 0]
]

locations = ["A", "B", "C", "D"]
n = len(locations)

# ACO parameters
num_ants = 4
iterations = 10
alpha = 1
beta = 2
evaporation = 0.5

# Initialize pheromone
pheromone = [[1.0 for j in range(n)] for i in range(n)]


def choose_next(current, unvisited):
    probabilities = []

    for city in unvisited:
        value = (pheromone[current][city] ** alpha) * \
                ((1 / distance[current][city]) ** beta)
        probabilities.append((city, value))

    total = sum(value for city, value in probabilities)

    r = random.random() * total
    cumulative = 0

    for city, value in probabilities:
        cumulative += value

        if r <= cumulative:
            return city


best_route = None
best_distance = float('inf')


for iteration in range(iterations):

    routes = []

    for ant in range(num_ants):

        start = random.randint(0, n - 1)

        route = [start]
        unvisited = list(range(n))
        unvisited.remove(start)

        while unvisited:
            current = route[-1]
            next_city = choose_next(current, unvisited)
            route.append(next_city)
            unvisited.remove(next_city)

        # Return to starting point
        route.append(start)

        total_distance = 0

        for i in range(len(route) - 1):
            total_distance += distance[route[i]][route[i + 1]]

        routes.append((route, total_distance))

        if total_distance < best_distance:
            best_distance = total_distance
            best_route = route.copy()

    # Pheromone evaporation
    for i in range(n):
        for j in range(n):
            pheromone[i][j] *= (1 - evaporation)

    # Pheromone update
    for route, total_distance in routes:

        deposit = 1 / total_distance

        for i in range(len(route) - 1):
            a = route[i]
            b = route[i + 1]

            pheromone[a][b] += deposit
            pheromone[b][a] += deposit


print("Best Route:")

for city in best_route:
    print(locations[city], end=" ")

print()

print("Minimum Total Trip Duration:", best_distance)

print("\nFinal Pheromone Values:")

for row in pheromone:
    print([round(value, 3) for value in row])
