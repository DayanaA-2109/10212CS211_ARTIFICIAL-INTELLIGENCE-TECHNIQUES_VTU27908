from queue import PriorityQueue

graph = {
    'A': [('B', 1), ('C', 3)],
    'B': [('D', 3), ('E', 6)],
    'C': [('F', 5)],
    'D': [('G', 2)],
    'E': [('G', 1)],
    'F': [('G', 2)],
    'G': []
}

heuristic = {
    'A': 7,
    'B': 6,
    'C': 4,
    'D': 2,
    'E': 1,
    'F': 2,
    'G': 0
}

def a_star(start, goal):
    open_list = PriorityQueue()
    open_list.put((heuristic[start], heuristic[start], start))

    g_cost = {start: 0}
    parent = {start: None}
    visited = set()

    while not open_list.empty():
        f, h, current = open_list.get()

        if current in visited:
            continue

        visited.add(current)

        if current == goal:
            break

        for neighbor, cost in graph[current]:
            new_cost = g_cost[current] + cost

            if neighbor not in g_cost or new_cost < g_cost[neighbor]:
                g_cost[neighbor] = new_cost
                parent[neighbor] = current

                f_cost = new_cost + heuristic[neighbor]
                open_list.put((f_cost, heuristic[neighbor], neighbor))

    # Construct path
    path = []
    node = goal

    while node is not None:
        path.append(node)
        node = parent[node]

    path.reverse()

    # Return path and total cost
    return path, g_cost[goal]


optimal_path, total_cost = a_star('A', 'G')

print("Optimal Path:", " → ".join(optimal_path))
print("Total Cost:", total_cost)


OUTPUT:

Optimal Path: A → B → D → G
Total Cost: 6
